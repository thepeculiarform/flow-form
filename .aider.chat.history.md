from Project import BaffleCenterline
import ghpythonlib.components as gh
import rhino3dm as rg

def get_baffle_data():
    """Get all baffles from Rhino and return as Polars DataFrame"""
    baffles = []
    for centerline in BaffleCenterline.get_all():
        for baffle in centerline.baffles:
            baffles.append(baffle.to_dataframe())
    return pl.DataFrame(baffles)

def main():
    st.title("Baffle System Configuration")
    
    # Get data from Rhino
    df = get_baffle_data()
    
    # Show raw data
    st.subheader("All Baffles")
